# mutate.tcl; STATIC-COMPILE.md, "Decision economics"): deliberate bugs in
# the batched scheduler (hir/static.tcl), its dependency analysis
# (hir/staticdeps.tcl) and the reuse of method-call decisions
# (hir/static.tcl's MethodSeeds, hir/hir.tcl). The first group changes what
# is decided (semantic); the second only how many builds it takes
# (economic: the milestone's contract is economic, so these are bugs too).

@@ contribution-not-discarded an erasing call inside a block is not a dependency on that block: rejecting the block keeps its contribution in the decisions' model
## hir/staticdeps.tcl
<<<<
                if {[dict exists $state siteOf $e]} {
                    dict set succ Z:[dict get $state siteOf $e] erasure
                }
                lassign [dict get $hir exprs $e target] tk tb
                set args [dict get $hir exprs $e args]
                if {$tk ne "block" || ![dict exists $hir exprs $tb]} {
                    foreach arg $args {
                        AddSubtree $hir state $arg succ erasure
                    }
                    continue
                }
                dict set succ R:$tb erasure
                foreach entry [hir::signatures::entryTypes $hir $tb] arg $args {
                    if {$entry eq "" && $arg ne ""} {
                        AddSubtree $hir state $arg succ erasure
                    }
                }
====
                lassign [dict get $hir exprs $e target] tk tb
                if {$tk eq "block"} {
                    dict set succ R:$tb erasure
                }
>>>>

                dict set decided $other $choice
            }
        }
    }
