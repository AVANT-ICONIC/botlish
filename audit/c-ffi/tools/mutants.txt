# Mutants of `extern c fn` and `extern c library` (audit/c-ffi/tools/
# mutate.tcl; C-FFI.md): each a deliberate bug in the declaration grammar
# (surface/parser.tcl), the registration and link plan (hir/foreign.tcl),
# the borrow admission (hir/borrow.tcl), the direct analysis (hir/direct.tcl),
# HIR interchange (hir/interchange.tcl), the Tcl refusal (core/cabi.tcl),
# native lowering (native/lower.tcl), the C call's code generation
# (native/src/codegen/clif.rs), its runtime helpers (native/src/runtime/
# ffi.rs) or the link step (native/src/codegen/aot.rs). The brief's mutants
# come first, then further ones.
#
#   @@ NAME DESCRIPTION
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# Lines starting with # outside an edit are comments.

@@ nearest-library each C function is associated with the last library its own unit declares, and only libraries some function is associated with are linked
## hir/foreign.tcl
<<<<
    set links $plan
}
====
    set near {}
    foreach decl $decls {
        set best ""
        foreach link $linkDecls {
            if {[dict get $link namespace] eq [dict get $decl namespace]} {
                set best [dict get $link library]
            }
        }
        if {$best ne "" && $best ni $near} {
            lappend near $best
        }
    }
    set links [lmap entry $plan {
        if {[dict get $entry library] ni $near} continue
        set entry
    }]
}

# --- the macOS Apple Silicon C ABI (MACOS-AARCH64-MILESTONE-6.md) ------------
#
# A `%% TARGET ...` line restricts a mutant to the hosts that compile it
# differently; elsewhere it is SKIPPED, never counted as killed. Not here,
# because they are equivalent: a C call that is no safepoint of the stack
# maps (no C function can reach the collector during the call, C-FFI.md
# section 20: no root is ever read there), and Linux AArch64's extension
# attributes (Cranelift ignores them for AAPCS64, whose callee extends).
