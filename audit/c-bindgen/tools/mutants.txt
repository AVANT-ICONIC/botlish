# Mutants of the C binding generator (audit/c-bindgen/tools/mutate.tcl;
# C-BINDGEN.md): each a deliberate bug in the binding model
# (tools/c-bindgen/model.tcl), the compiler probes (probe.tcl), the emitter
# (emit.tcl), the report and manifest (report.tcl), the profile reader
# (profile.tcl) or the pipeline (generate.tcl). The brief's mutants come
# first, then further ones.
#
#   @@ NAME DESCRIPTION
#   ## FILE
#   <<<<
#   text to replace (must occur exactly once in FILE)
#   ====
#   replacement
#   >>>>
#
# Lines starting with # outside an edit are comments.

@@ const-noescape a pointer to const is assumed not to escape
## tools/c-bindgen/model.tcl
<<<<

        dict set p escape $escape
====
        set pt [Unq [dict get $p ctype]]
        if {$escape eq "" && [lindex $pt 0] eq "ptr" && "const" in [::cir::quals [lindex $pt 1]]} {
            set escape declared-noescape
        }
        dict set p access $access
        dict set p escape $escape
